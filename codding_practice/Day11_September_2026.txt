## jenkins Pipeline - Email notifications

you will learn how to  send email notifications on failur or success

pipeline {
    agent any
    enviroment {
        INSTITUTE_NAME = "Pathnex"
    }

    stage {
        stage('Checkout') {
            steps {
                git url: 'https://github.com/Pathnex/sample-java-app.git'
            }
        }

        stage('Build') {
            step {
                steps {
                    sh 'mvn clean package'
                }
            }
        }
    }
    post {
        success {
            mail to: 'myemail@pathnex.com',
                subject: "SUCCESS: Build #${env.BUILD_NUMBER}"
                body: "build complated successfully for $INSTITUTE_NAME"
        }
        failur {
            mail to: 'team@pathnex.com'
                subject: "FAILURE: Build #${env.BUILD_NUMBER}"
                body: "Build failed for $INSTITUTE_NAME"
        }
    }
}
