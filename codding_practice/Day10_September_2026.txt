## Terraform - Route Table + Association

resource "aws_route_table" "pathnexRT" {
    vpc_id = aws_vpc.Pathnex.vpc_id
    route {
        cidr_block = "0.0.0.0/0"
        geteway_id = aws_internet_geteway.PathnexIGW.vpc_id
    }
    tags = {
        Name = "Pathnex-RouteTable"
    }
}

resource "aws_route_table_association" "PathnexRTA" {
    sunet_id = aws_subnet.pathnexSunet.vpc_id
    route_table_id = aws_route_table.pathnex.id 
}

## caching Dependencies

you will learn how to reuse dependencies to speed up builds

pipeline {
    aget any
    enviroment {
        INSTITUTE_NAME = "Pathnex"
    }
    stages {
        stage("Checkout") {
            steps {
                git url: 'myapprepo'
            }
        }

        stage('Build') {
            steps {
                sh '''
                if[-d ".m"]; then
                    echo "Using cached dependencies"
                else
                    mvn clean compile
                fi
                '''
            }
        }

    }
}
