## Terraform security group r6i4xlarge EC2
 provider "aws" {
	region = "us-east-1"
}

 resource "aws_security_group" "pathnexSG" {
  	name = "pathnextSG"
	description = "Pathnex SG for SSH"

	ingress {
		from_port = 22
		to_port   = 22
		protocol = 'tcp'
		cidr_locks = [0.0.0.0/0]
		}
	}
resource "aws_instance" "PathnexEc2" {
	ami = "ami-myaminum"
	instance_type ="r6i.4xlarge"
	vpc_security_group_ides = [aws_security_group.pathnex.id]
	
	tags = {
		name= 'pathnex-EC2'
		}
	}

# Node Build
## jenkins pipline nodejs build
pipline {
	agent any
	tools {
		nodejs 'NodeJS-16'
	}
	stage {
	stage ('Checkout') {
		step {
			git url:"mygiturl"
		}
		stage ('Install Dependincies') {
			step {
				sh 'npm install'
			}
		stage('Run Tests'){
			step {
				sh 'npm run build'
			}
		}
	}
}
